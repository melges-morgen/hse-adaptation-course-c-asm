\begin{thebibliography}{99}
\addcontentsline{toc}{chapter}{Используемая литература}

\bibitem{kernighan-ritchie-c}
Керниган Б., Ритчи Д. \textit{Язык программирования C}. Классический источник по языку C и его модели программирования.

\bibitem{bryant-ohallaron-csapp}
Bryant R. E., O'Hallaron D. R. \textit{Computer Systems: A Programmer's Perspective}. Базовый источник о связи программ, памяти, процессора, компиляции и операционной системы.

\bibitem{patterson-hennessy}
Patterson D. A., Hennessy J. L. \textit{Computer Organization and Design}. Введение в архитектуру компьютеров и машинное представление данных.

\bibitem{intel-manual}
Intel. \textit{Intel 64 and IA-32 Architectures Software Developer's Manual}. Справочная документация по архитектуре x86-64.

\bibitem{nasm-doc}
The NASM Development Team. \textit{NASM Documentation}. Справочная документация по ассемблеру NASM.

\bibitem{gcc-manual}
Free Software Foundation. \textit{Using the GNU Compiler Collection}. Документация по GCC и этапам компиляции.

\bibitem{gnu-make-manual}
Free Software Foundation. \textit{GNU Make Manual}. Документация по системе сборки \texttt{make}.

\bibitem{pro-git}
Chacon S., Straub B. \textit{Pro Git}. Практическое введение в Git и командную разработку.

\bibitem{linux-man-pages}
The Linux man-pages project. \textit{Linux Programmer's Manual}. Справочные страницы Linux для системных вызовов, библиотечных функций и команд.

\bibitem{os-three-easy-pieces}
Arpaci-Dusseau R. H., Arpaci-Dusseau A. C. \textit{Operating Systems: Three Easy Pieces}.
Введение в процессы, виртуальную память, конкуренцию и постоянное хранение.
\url{https://pages.cs.wisc.edu/~remzi/OSTEP/}

\bibitem{ritchie-unix-history}
Ritchie D. M. \textit{The Evolution of the Unix Time-Sharing System}.
In: \textit{Language Design and Programming Methodology}, Lecture Notes in
Computer Science 79, 1980, pp. 25--35. DOI: \url{https://doi.org/10.1007/3-540-09745-7_2}.

\bibitem{ritchie-thompson-unix}
Ritchie D. M., Thompson K. \textit{The UNIX Time-Sharing System}.
Communications of the ACM, vol. 17, no. 7, 1974, pp. 365--375.
DOI: \url{https://doi.org/10.1145/361011.361061}.

\bibitem{bourne-unix-shell}
Bourne S. R. \textit{The UNIX Time-Sharing System: The UNIX Shell}.
Bell System Technical Journal, vol. 57, no. 6, 1978, pp. 1971--1990.
DOI: \url{https://doi.org/10.1002/j.1538-7305.1978.tb02139.x}.

\bibitem{ibm-system-360-history}
IBM. \textit{IBM System/360}. Историческая справка об объявлении семейства System/360 в 1964 году.
\url{https://www.ibm.com/history/system-360}

\bibitem{ibm-jcl-reference}
IBM. \textit{z/OS MVS JCL User's Guide}. Справочник по управляющему языку заданий, включая условное выполнение шагов по их return codes.

\bibitem{brooks-mythical-man-month}
Brooks F. P. \textit{The Mythical Man-Month: Essays on Software Engineering}.
Воспоминания о разработке OS/360 и организации большого программного проекта.

\bibitem{goldberg-floating-point}
Goldberg D. \textit{What Every Computer Scientist Should Know About Floating-Point Arithmetic}. Классическая статья о числах с плавающей точкой и ошибках округления.

\bibitem{ieee-754}
IEEE. \textit{IEEE Standard for Floating-Point Arithmetic}, IEEE 754-2019.
Стандарт представления и арифметики чисел с плавающей точкой.

\bibitem{unicode-standard}
The Unicode Consortium. \textit{The Unicode Standard}, глава 3:
кодовые точки, скалярные значения, UTF-8 и UTF-16.
\url{https://www.unicode.org/versions/latest/}

\bibitem{rfc3629}
Yergeau F. \textit{UTF-8, a transformation format of ISO 10646}. RFC 3629.
\url{https://www.rfc-editor.org/rfc/rfc3629}

\bibitem{c11-draft}
ISO/IEC JTC1/SC22/WG14. \textit{N1570: Committee Draft ISO/IEC 9899:2011}.
Открытый проект стандарта C: целочисленные преобразования, арифметика,
сдвиги и строки.
\url{https://www.open-std.org/jtc1/sc22/wg14/www/docs/n1570.pdf}

\bibitem{amd64-psabi}
System V Application Binary Interface. \textit{AMD64 Architecture Processor
Supplement}. Регистры аргументов, сохраняемые регистры и модель LP64.
\url{https://gitlab.com/x86-psABIs/x86-64-ABI}
